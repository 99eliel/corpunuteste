rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function logado() {
      return request.auth != null;
    }

    function usuarioExiste() {
      return logado()
        && exists(/databases/$(database)/documents/usuarios/$(request.auth.uid));
    }

    function meuPerfil() {
      return get(/databases/$(database)/documents/usuarios/$(request.auth.uid));
    }

    function ativo() {
      return usuarioExiste()
        && meuPerfil().data.ativo == true;
    }

    function admin() {
      return ativo()
        && meuPerfil().data.tipo == "admin";
    }

    function usuarioComum() {
      return ativo();
    }

    function financeiroPodeDefinirValor() {
      return admin() || (
        ativo()
        && meuPerfil().data.keys().hasAny(["permissoes"])
        && meuPerfil().data.permissoes.keys().hasAny(["recursos"])
        && (
          (
            meuPerfil().data.permissoes.recursos.keys().hasAny(["gerenciarValores"])
            && meuPerfil().data.permissoes.recursos.gerenciarValores == true
          )
          || (
            meuPerfil().data.permissoes.recursos.keys().hasAny(["marcarPagamentos"])
            && meuPerfil().data.permissoes.recursos.marcarPagamentos == true
          )
        )
      );
    }

    function alteracaoValorTotalManualPermitida() {
      return financeiroPodeDefinirValor()
        && resource.data.statusPagamento != "pago"
        && request.resource.data.statusPagamento == "pendente"
        && request.resource.data.atualizadoPor == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          "total",
          "subtotal",
          "valorUnitario",
          "statusPagamento",
          "valorPendente",
          "valorManualFinanceiroPendente",
          "valorManualFinanceiro",
          "valorTotalDefinidoManualmente",
          "valorTotalManual",
          "formaValorPagamento",
          "motivoValorPendente",
          "avisoPagamento",
          "valorInformadoPor",
          "valorInformadoEm",
          "atualizadoPor",
          "atualizadoEm",
          "versaoValorManual",
          "observacoes"
        ]);
    }

    function dadosPrecoPadraoAlcaValidos() {
      return request.resource.data.referencia == "TODAS"
        && request.resource.data.processo == "ALÇA"
        && request.resource.data.setor == "alca"
        && request.resource.data.setorLabel == "Alça"
        && request.resource.data.valor is number
        && request.resource.data.valor > 0
        && request.resource.data.ativo == true
        && request.resource.data.tipoValor == "padrao_global_alca"
        && request.resource.data.valorPadraoGlobalAlca == true
        && request.resource.data.multiplicadorQuantidade == 2
        && request.resource.data.atualizadoPor == request.auth.uid;
    }

    function novoPrecoPadraoAlcaPermitido(precoId) {
      return precoId == "valor-padrao-alca"
        && financeiroPodeDefinirValor()
        && dadosPrecoPadraoAlcaValidos()
        && request.resource.data.criadoPor == request.auth.uid
        && request.resource.data.keys().hasOnly([
          "referencia",
          "processo",
          "setor",
          "setorLabel",
          "valor",
          "ativo",
          "tipoValor",
          "valorPadraoGlobalAlca",
          "multiplicadorQuantidade",
          "atualizadoPor",
          "atualizadoEm",
          "versaoValorAlca",
          "criadoPor",
          "criadoEm"
        ]);
    }

    function alteracaoPrecoPadraoAlcaPermitida(precoId) {
      return precoId == "valor-padrao-alca"
        && financeiroPodeDefinirValor()
        && dadosPrecoPadraoAlcaValidos()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          "referencia",
          "processo",
          "setor",
          "setorLabel",
          "valor",
          "ativo",
          "tipoValor",
          "valorPadraoGlobalAlca",
          "multiplicadorQuantidade",
          "atualizadoPor",
          "atualizadoEm",
          "versaoValorAlca"
        ]);
    }

    function pagamentoEhAlca() {
      return (
        resource.data.processo == "ALÇA"
        || resource.data.processo == "ALCA"
        || resource.data.servicoNome == "ALÇA"
        || resource.data.servicoNome == "ALCA"
        || resource.data.processoMovimentacao == "ALÇA"
        || resource.data.processoMovimentacao == "ALCA"
      );
    }

    function alteracaoPagamentoAlcaPermitida() {
      return financeiroPodeDefinirValor()
        && pagamentoEhAlca()
        && resource.data.statusPagamento != "pago"
        && request.resource.data.statusPagamento == "pendente"
        && request.resource.data.atualizadoPor == request.auth.uid
        && request.resource.data.precoReferenciaId == "valor-padrao-alca"
        && request.resource.data.servicoId == "valor-padrao-alca"
        && request.resource.data.multiplicadorAlcas == 2
        && request.resource.data.valorUnitarioAlca is number
        && request.resource.data.valorUnitarioAlca > 0
        && request.resource.data.valorUnitario is number
        && request.resource.data.valorUnitario > 0
        && request.resource.data.quantidadeAlcas is number
        && request.resource.data.quantidadeAlcas == request.resource.data.quantidade * 2
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          "precoReferenciaId",
          "servicoId",
          "quantidadeAlcas",
          "multiplicadorAlcas",
          "valorUnitarioAlca",
          "valorUnitario",
          "subtotal",
          "total",
          "statusPagamento",
          "valorPendente",
          "valorManualFinanceiroPendente",
          "formaValorPagamento",
          "motivoValorPendente",
          "avisoPagamento",
          "observacoes",
          "atualizadoPor",
          "atualizadoEm",
          "versaoValorAlca"
        ]);
    }

    function pagamentoProcessoRevisaoSutia() {
      return (
        resource.data.keys().hasAny(["processo"])
        && resource.data.processo in [
          "SUTIÃ MONTAGEM",
          "SUTIA MONTAGEM",
          "SUTIÃ COMPLETO",
          "SUTIA COMPLETO"
        ]
      ) || (
        resource.data.keys().hasAny(["servicoNome"])
        && resource.data.servicoNome in [
          "SUTIÃ MONTAGEM",
          "SUTIA MONTAGEM",
          "SUTIÃ COMPLETO",
          "SUTIA COMPLETO"
        ]
      ) || (
        resource.data.keys().hasAny(["processoMovimentacao"])
        && resource.data.processoMovimentacao in [
          "SUTIÃ MONTAGEM",
          "SUTIA MONTAGEM",
          "SUTIÃ COMPLETO",
          "SUTIA COMPLETO"
        ]
      );
    }

    function alteracaoRevisaoComponentesPermitida() {
      return usuarioComum()
        && pagamentoProcessoRevisaoSutia()
        && resource.data.statusPagamento != "pago"
        && request.resource.data.statusPagamento == resource.data.statusPagamento
        && request.resource.data.atualizadoPor == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          "valorUnitario",
          "subtotal",
          "total",
          "valorUnitarioBaseRevisao",
          "valorTotalBaseRevisao",
          "revisaoComponentesAplicada",
          "revisaoLateralAtiva",
          "revisaoBojoAtiva",
          "descontoRevisaoLateralUnitario",
          "descontoRevisaoBojoUnitario",
          "descontoRevisaoComponentesUnitario",
          "descontoRevisaoComponentesTotal",
          "revisaoComponentesTotalCalculado",
          "revisaoComponentesValorUnitarioCalculado",
          "revisaoComponentesVersao",
          "revisaoComponentesAtualizadaEm",
          "atualizadoPor",
          "atualizadoEm"
        ]);
    }

    function pagamentoDoUsuario() {
      return usuarioComum()
        && resource.data.criadoPor == request.auth.uid;
    }

    function novoPagamentoDoUsuario() {
      return usuarioComum()
        && request.resource.data.criadoPor == request.auth.uid
        && request.resource.data.atualizadoPor == request.auth.uid;
    }

    function novoLogDoUsuario() {
      return usuarioComum() && (
        (
          request.resource.data.keys().hasAny(["usuarioUid"])
          && request.resource.data.usuarioUid == request.auth.uid
        )
        || (
          request.resource.data.keys().hasAny(["usuarioId"])
          && request.resource.data.usuarioId == request.auth.uid
        )
        || (
          request.resource.data.keys().hasAny(["criadoPor"])
          && request.resource.data.criadoPor == request.auth.uid
        )
      );
    }

    match /usuarios/{userId} {
      allow read: if admin() || (logado() && request.auth.uid == userId);
      allow create: if admin();
      allow update: if admin();
      allow delete: if admin();
    }

    match /produtos/{produtoId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /ordensProducao/{ordemId} {
      allow read: if usuarioComum();
      allow create, update: if usuarioComum();
      allow delete: if admin();
    }

    match /faccoes/{faccaoId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /manejos/{manejoId} {
      allow read: if usuarioComum();
      allow create, update: if usuarioComum();
      allow delete: if admin();
    }

    match /fasesManejo/{faseId} {
      allow read: if usuarioComum();
      allow create, update: if usuarioComum();
      allow delete: if admin();
    }

    match /logsAlteracoes/{logId} {
      allow read: if admin();
      allow create: if novoLogDoUsuario();
      allow update, delete: if false;
    }

    match /configuracoes/{docId} {
      allow read: if usuarioComum();
      allow write: if admin();
    }

    match /celulas/{celulaId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /movimentacoesProducao/{movimentoId} {
      allow read: if usuarioComum();
      allow create, update: if usuarioComum();
      allow delete: if admin();
    }

    match /precosReferencia/{precoId} {
      allow read: if usuarioComum();
      allow create: if admin() || novoPrecoPadraoAlcaPermitido(precoId);
      allow update: if admin() || alteracaoPrecoPadraoAlcaPermitida(precoId);
      allow delete: if admin();
    }

    match /entregasPagamento/{entregaId} {
      allow read: if usuarioComum();
      allow create: if admin() || novoPagamentoDoUsuario();
      allow update: if admin()
        || alteracaoValorTotalManualPermitida()
        || alteracaoPagamentoAlcaPermitida()
        || alteracaoRevisaoComponentesPermitida()
        || (
          pagamentoDoUsuario()
          && resource.data.statusPagamento != "pago"
          && novoPagamentoDoUsuario()
          && request.resource.data.statusPagamento != "pago"
        );
      allow delete: if admin() || (
        pagamentoDoUsuario()
        && resource.data.statusPagamento != "pago"
      );
    }

    match /ajustesFinanceirosFaccoes/{ajusteId} {
      allow read: if financeiroPodeDefinirValor();

      allow create: if financeiroPodeDefinirValor()
        && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.faccao is string
        && request.resource.data.faccao.size() > 0
        && request.resource.data.tipo in ["faccao_deve_empresa", "empresa_deve_faccao"]
        && request.resource.data.valor is number
        && request.resource.data.valor > 0
        && request.resource.data.status == "pendente";

      allow update: if financeiroPodeDefinirValor()
        && request.resource.data.updatedBy == request.auth.uid
        && request.resource.data.status in ["pendente", "quitado"]
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          "status",
          "updatedAt",
          "updatedBy",
          "quitadoAt",
          "quitadoPor"
        ]);

      allow delete: if financeiroPodeDefinirValor()
        && resource.data.status == "pendente";
    }

    match /ajustesMigracao/{ajusteId} {
      allow read: if admin() || (
        usuarioComum()
        && resource.data.criadoPor == request.auth.uid
      );
      allow create: if admin() || (
        usuarioComum()
        && request.resource.data.criadoPor == request.auth.uid
      );
      allow update, delete: if admin();
    }

    match /relatoriosMigracaoLigia/{relatorioId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /datasIncoerentesLigia/{dataId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /processosMigracao/{processoId} {
      allow read: if usuarioComum();
      allow create, update, delete: if admin();
    }

    match /travasOperacionais/{travaId} {
      allow read: if usuarioComum();
      allow create: if usuarioComum()
        && request.resource.data.criadoPor == request.auth.uid;
      allow update: if usuarioComum()
        && request.resource.data.criadoPor == request.auth.uid;
      allow delete: if admin() || (
        usuarioComum()
        && resource.data.criadoPor == request.auth.uid
      );
    }
  }
}
